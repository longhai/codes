  <script>
    let alarm = localStorage.getItem('savedAlarmTime');
    let alarmNote = localStorage.getItem('savedAlarmNote') || '';
    let lang = localStorage.getItem('savedLang') || 'vi-VN';
    lucide.createIcons();

    const tts = {
      'vi-VN': { a:(h,m,n)=>`Bây giờ là ${h} giờ ${m} phút. ${n ? 'Lời nhắc: ' + n : 'Dậy đi thôi!'}`, t:(h,m)=>`Bây giờ là ${h} giờ ${m} phút` },
      'en-US': { a:(h,m,n)=>`It is ${h}:${m<10?'0'+m:m}. ${n ? 'Reminder: ' + n : 'Wake up!'}`, t:(h,m)=>`It is ${h}:${m<10?'0'+m:m}` },
      'ja-JP': { a:(h,m,n)=>`${h}時${m}分です。${n ? 'リマインダー: ' + n : '時間です！'}`, t:(h,m)=>`現在は${h}時${m}分です` },
      'ko-KR': { a:(h,m,n)=>`${h}시 ${m}분입니다. ${n ? '알림: ' + n : '일어나세요!'}`, t:(h,m)=>`지금은 ${h}시 ${m}분입니다` },
      'zh-CN': { a:(h,m,n)=>`现在是${h}点${m}分。${n ? '提醒：' + n : '起床了！'}`, t:(h,m)=>`现在是${h}点${m}分` },
      'fr-FR': { a:(h,m,n)=>`Il est ${h} heures ${m}. ${n ? 'Rappel: ' + n : 'Réveillez-vous!'}`, t:(h,m)=>`Il est ${h} heures ${m}` },
      'de-DE': { a:(h,m,n)=>`Es ist ${h} Uhr ${m}. ${n ? 'Erinnerung: ' + n : 'Aufstehen!'}`, t:(h,m)=>`Es ist ${h} Uhr ${m}` }
    };

    document.addEventListener('click', () => { if('speechSynthesis' in window) window.speechSynthesis.resume(); }, { once: true });

    window.onload = () => {
      document.getElementById('langSelect').value = lang;
      updateUI();
      document.getElementById('clockBtn').onclick = speakTime;
      document.getElementById('alarmBar').onclick = () => toggleModal(true);
    };

    setInterval(() => {
      const d = new Date();
      document.getElementById('date').innerText = d.toLocaleDateString(lang, { weekday:'short', day:'2-digit', month:'short', year:'numeric' });

      let h = d.getHours(), m = String(d.getMinutes()).padStart(2,'0'), s = String(d.getSeconds()).padStart(2,'0');
      document.getElementById('time').innerText = `${String(h%12||12).padStart(2,'0')}:${m}:${s}`;
      document.getElementById('ampm').innerText = h >= 12 ? 'PM' : 'AM';

      if (alarm === `${String(h).padStart(2,'0')}:${m}` && s === '00') {
        speak(tts[lang].a(h, d.getMinutes(), alarmNote));
      }
    }, 1000);

    function toggleModal(open) {
      if(open) {
        if(alarm) document.getElementById('alarmInput').value = alarm;
        document.getElementById('noteInput').value = alarmNote;
      }
      document.getElementById('modal').classList.toggle('open', open);
    }

    function saveAlarm() {
      const v = document.getElementById('alarmInput').value;
      if (v) {
        alarm = v;
        alarmNote = document.getElementById('noteInput').value.trim();
        localStorage.setItem('savedAlarmTime', v);
        localStorage.setItem('savedAlarmNote', alarmNote);
        updateUI();
        toggleModal(false);
      }
    }

    function turnOffAlarm() {
      alarm = null; alarmNote = '';
      localStorage.removeItem('savedAlarmTime');
      localStorage.removeItem('savedAlarmNote');
      updateUI();
      toggleModal(false);
    }

    function saveLang() {
      lang = document.getElementById('langSelect').value;
      localStorage.setItem('savedLang', lang);
    }

    function updateUI() {
      const ab = document.getElementById('alarmBar'), off = document.getElementById('btnOff'), noteEl = document.getElementById('alarmNoteText');
      if (alarm) {
        ab.classList.add('active');
        let [h, m] = alarm.split(':'), h12 = parseInt(h);
        document.getElementById('alarmText').innerText = `${String(h12%12||12).padStart(2,'0')}:${m} ${h12>=12?'PM':'AM'}`;
        noteEl.innerText = alarmNote ? `• ${alarmNote}` : '';
        off.style.display = 'block';
      } else {
        ab.classList.remove('active');
        document.getElementById('alarmText').innerText = '--:--';
        noteEl.innerText = '';
        off.style.display = 'none';
      }
    }

    function speak(txt) {
      if (!('speechSynthesis' in window)) return;
      window.speechSynthesis.cancel();
      const u = new SpeechSynthesisUtterance(txt); u.lang = lang;
      window.speechSynthesis.speak(u);
    }

    function speakTime() { const d = new Date(); speak(tts[lang].t(d.getHours(), d.getMinutes())); }
  </script>
